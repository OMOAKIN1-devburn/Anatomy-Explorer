Part 1: remember where the visitor stopped

This saves the animal, the tab, the labels on or off, and the selected dot in the visitor’s own browser. It’s remembered on the same phone or browser only.

Step 1. Make a new folder src/hooks. Right-click src, choose New Folder, and name it hooks. Inside it, create usePersistentState.ts:

import { useEffect, useState } from 'react';

/** Works like useState, but remembers the value in this browser between visits. */
export function usePersistentState<T>(key: string, initial: T) {
  const [value, setValue] = useState<T>(() => {
    try {
      const saved = window.localStorage.getItem(key);
      return saved !== null ? (JSON.parse(saved) as T) : initial;
    } catch {
      return initial;
    }
  });

  useEffect(() => {
    try {
      window.localStorage.setItem(key, JSON.stringify(value));
    } catch {
      // Storage can be blocked (private mode). The page still works without it.
    }
  }, [key, value]);

  return [value, setValue] as const;
}

It opens automatically on a first visit and closes with a “Got it” button. After that, the ? button in the header reopens it. The wording is short and plain.

New file src/components/HowToDialog.tsx:

import { useEffect } from 'react';

const STEPS = [
  { icon: 'fa-hand-pointer', title: 'Choose a specimen', text: 'Pick an animal, or the human, from the collection.' },
  { icon: 'fa-arrows-rotate', title: 'Look around', text: 'Drag to rotate. Use the + and − buttons, or hold Ctrl and scroll, to zoom.' },
  { icon: 'fa-tags', title: 'Show the labels', text: 'Press “Click here to see the labels” above the specimen. Dots appear on the body parts.' },
  { icon: 'fa-book-open', title: 'Learn', text: 'Tap a dot to read about that body part. The profile shows habitat, diet and facts.' },
] as const;

export function HowToDialog({ open, onClose }: { open: boolean; onClose: () => void }) {
  useEffect(() => {
    if (!open) return;
    const onKey = (event: KeyboardEvent) => {
      if (event.key === 'Escape') onClose();
    };
    window.addEventListener('keydown', onKey);
    return () => window.removeEventListener('keydown', onKey);
  }, [open, onClose]);

  if (!open) return null;

  return (
    <div className="fixed inset-0 z-50 grid place-items-center bg-black/70 px-4" onClick={onClose}>
      <div
        role="dialog"
        aria-modal="true"
        aria-labelledby="howto-title"
        onClick={(event) => event.stopPropagation()}
        className="panel max-h-[90dvh] w-full max-w-lg overflow-y-auto p-6"
      >
        <p className="font-mono text-xs uppercase tracking-[.16em] text-[#e5ad59]">Quick guide</p>
        <h2 id="howto-title" className="font-display mt-1 text-2xl text-[#f0e8d8]">How to use the museum</h2>
        <ol className="mt-5 space-y-4">
          {STEPS.map((step, index) => (
            <li key={step.title} className="flex gap-4">
              <span className="grid h-10 w-10 shrink-0 place-items-center rounded-full border border-[#e5ad59]/50 bg-[#111617] text-[#e5ad59]">
                <i className={`fa-solid ${step.icon}`} aria-hidden="true" />
              </span>
              <div>
                <p className="text-base font-semibold text-[#f0e8d8]">{index + 1}. {step.title}</p>
                <p className="mt-0.5 text-sm leading-6 text-[#cfc8b8]">{step.text}</p>
              </div>
            </li>
          ))}
        </ol>
        <p className="mt-5 text-sm text-[#a9ada8]">Your place is saved, so you can continue where you left off.</p>
        <button
          type="button"
          onClick={onClose}
          autoFocus
          className="mt-5 w-full rounded-full border border-[#e5ad59]/60 bg-[#111617]/90 px-5 py-3 text-sm font-semibold text-[#f0dcae] hover:bg-[#1c2326]"
        >
          Got it
        </button>
      </div>
    </div>
  );
}
